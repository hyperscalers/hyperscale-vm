add_liquidity(bucket, bucket) — fallible
  takes    arg0 carrying config.x
  takes    arg1 carrying config.y
  mints    "Share" fungible
  declares
      0  credit-only self.reserves[config.x]
      1  credit-only self.reserves[config.y]
      2  write self.supply
  yields   edge 0 carrying self-issued(fungible["Share"])
  exports  handles for clauses 0–2, edge arg0, edge arg1

instantiate() — fallible
  declares
      0  requires self.config is absent
      1  write self.config
      2  requires self.resource[self-issued(fungible["Share"])] is absent
      3  write self.resource[self-issued(fungible["Share"])]
  exports  handle for clause 1, handle for clause 3, self.record, config.fee, config.x, config.y

remove_liquidity(bucket) — fallible
  takes    arg0 carrying self-issued(fungible["Share"])
  burns    "Share" fungible
  declares
      0  debit-only self.reserves[config.x]
      1  debit-only self.reserves[config.y]
      2  write self.supply
  yields   edge 0 carrying config.x
  yields   edge 1 carrying config.y
  exports  handles for clauses 0–2, edge arg0

swap(bucket, u128) — fallible
  takes    arg0 carrying (if resource-of(arg0) == config.x then config.x else config.y)
  declares
      0  credit-only self.reserves[if resource-of(arg0) == config.x then config.x else config.y]
      1  debit-only self.reserves[if resource-of(arg0) == config.x then config.y else config.x]
  yields   edge 0 carrying (if resource-of(arg0) == config.x then config.y else config.x)
  exports  handle for clause 0, handle for clause 1, config.fee, edge arg0, arg1
